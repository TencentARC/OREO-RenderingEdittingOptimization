#    # Install first PyTorch (official cu124 source)
#    pip install torch==2.6.0+cu124 torchvision==0.21.0+cu124 \
#      --index-url https://download.pytorch.org/whl/cu124
#
#    # Install again3DRelated dependencies (order matters)
#    pip install spconv-cu124==2.3.8  # will automatically install the matching cumm-cu124
#    pip install kaolin==0.18.0 -f https://nvidia-kaolin.s3.us-east-2.amazonaws.com/torch-2.6.0_cu124.html
#
#    # Install flash-attn (If you need a custom schema, Modify TORCH_CUDA_ARCH_LIST)
#    export TORCH_CUDA_ARCH_LIST="80;86;89;90"
#    pip install --no-build-isolation flash-attn==2.7.3
#    # flash-attn The special build declaration above has been met before installing the remaining packages.
#    pip install -r requirements.txt
#
# 3. Environment variable configuration (key):
#    # If you need to explicitly switch backends (Use it directly by default flash-attn can be ignored when)
