# ============================================================================
# 安装说明和注意事项
# ============================================================================
#
# 1. 环境要求:
#    - Python 3.12
#    - CUDA 12.4 (与 PyTorch 2.6.0+cu124 及 flash-attn 2.7.3 验证)
#    - 推荐使用 uv 环境管理
#
# 2. 安装顺序 (重要):
#    conda create -n grpo3d python=3.12
#    conda activate grpo3d
#    
#    # 先安装 PyTorch (官方 cu124 源)
#    pip install torch==2.6.0+cu124 torchvision==0.21.0+cu124 \
#      --index-url https://download.pytorch.org/whl/cu124
#    
#    # 再安装3D相关依赖 (顺序重要)
#    pip install spconv-cu124==2.3.8  # 将自动安装匹配的 cumm-cu124
#    pip install kaolin==0.18.0 -f https://nvidia-kaolin.s3.us-east-2.amazonaws.com/torch-2.6.0_cu124.html
#    
#    # 安装 flash-attn (如需自定义架构, 修改 TORCH_CUDA_ARCH_LIST)
#    export TORCH_CUDA_ARCH_LIST="80;86;89;90"
#    pip install --no-build-isolation flash-attn==2.7.3
#    # flash-attn 已满足上方的特殊构建声明，再安装其余包
#    pip install -r requirements.txt
#
# 3. 环境变量配置 (关键):
#    # 如需显式切换后端 (默认直接用 flash-attn 时可忽略)
#    export ATTN_BACKEND=flash_attn
#    export SPCONV_ALGO=auto          # 允许 spconv 自动选择 NVRTC，规避大 GEMM 的 int32 上限
#    export HF_HUB_OFFLINE=1          # 使用本地模型
#    export CUDA_VISIBLE_DEVICES=0    # 指定GPU
#
# 4. 使用建议:
#    # 推荐使用 SETUP.md 中的训练和评估脚本
#
# 5. 已知问题:
#    - 不要将 transformers 升级到 5.x；diffusers 0.38.0 的 Qwen prompt encoding 与其不兼容
#    - flash-attn 必须针对当前 torch 2.6.0+cu124 环境编译
#    - kaolin 的 wheel 索引必须与目标 torch 2.6.0 / CUDA 12.4 对齐
#
# 6. 验证安装:
#    python -c "import torch, flash_attn, spconv, kaolin; print('✅ 核心依赖安装成功')"
